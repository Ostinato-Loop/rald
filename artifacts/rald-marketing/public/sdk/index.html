import { rald } from '@rald/sdk'

// One client. Every service.
const client = new rald({
  apiKey: 'rald_live_...',
  environment: 'production'
})

// Verify a BVN in 2 seconds
const identity = await client.identity.verifyBVN({
  bvn: '12345678901',
  phone: '+2348012345678'
})

// Create a payment instantly
const payment = await client.payments.create({
  amount: 50000,
  currency: 'NGN',
  recipient: 'user@example.com'
})
